fcitx5-diagnose

# 환경변수 확인
env | grep -i im_module

# 프로세스 확인
pgrep -af fcitx5

# 설정 확인
cat ~/.config/fcitx5/profile
cat ~/.config/fcitx5/config

# 강제 재시작
fcitx5 -r
